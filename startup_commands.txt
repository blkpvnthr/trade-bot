Manage the existing service with:

# status / health
launchctl list com.quantumquant.trader925

# restart (pick it up after any change)
launchctl kickstart -k gui/$(id -u)/com.quantumquant.trader925

# stop (unload) / start (load)
launchctl bootout  gui/$(id -u)/com.quantumquant.trader925
launchctl bootstrap gui/$(id -u) ~/Library/LaunchAgents/com.quantumquant.trader925.plist
